# AnalyticsQueryRequest

## Example Usage

```typescript
import { AnalyticsQueryRequest } from "@meetkai/mka1/models/components";

let value: AnalyticsQueryRequest = {
  metric: "unique_visitors",
  from: new Date("2026-09-01T00:00:00Z"),
  to: new Date("2026-09-02T00:00:00Z"),
  eventName: "page_view",
  bucket: "hour",
  timezone: "America/Los_Angeles",
  groupBy: "pathname",
  filters: [
    {
      property: "plan",
      op: "eq",
      value: "pro",
    },
  ],
};
```
